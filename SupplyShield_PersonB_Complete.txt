import numpy as np

train = pd.read_csv("../data/datasplit/train_legs_v1.csv")
cal = pd.read_csv("../data/datasplit/calibration_legs_v1.csv")
test = pd.read_csv("../data/datasplit/test_legs_v1.csv")

def evaluate_baseline(data, predictions):

    actual_ratio = data["delay_ratio"].to_numpy()

    actual_excess = (
        (data["delay_ratio"] - 1) * data["osrm_time_min"]
    ).to_numpy()

    predicted_excess = (
        (predictions - 1) * data["osrm_time_min"]
    )

    return {
        "MAE_ratio": float(
            np.abs(actual_ratio - predictions).mean()
        ),
        "MedAE_ratio": float(
            np.median(np.abs(actual_ratio - predictions))
        ),
        "MAE_excess_min": float(
            np.abs(actual_excess - predicted_excess).mean()
        )
    }


# Baseline 1: Global median
global_median = float(train["delay_ratio"].median())

# Baseline 2: Lane median, calculated using training data only
lane_medians = train.groupby(
    ["source_center", "destination_center"]
)["delay_ratio"].median()

def predict_lane(data):

    predictions = []

    for source, destination in zip(
        data["source_center"],
        data["destination_center"]
    ):
        predictions.append(
            lane_medians.get(
                (source, destination),
                global_median
            )
        )

    return np.array(predictions)


# Evaluate both baselines
results = {}

for name, data in [
    ("calibration", cal),
    ("test", test)
]:

    global_predictions = np.full(
        len(data), global_median
    )

    lane_predictions = predict_lane(data)

    results[name] = {
        "global_median": evaluate_baseline(
            data, global_predictions
        ),
        "lane_median": evaluate_baseline(
            data, lane_predictions
        )
    }

print("Training global median:", global_median)

display(
    pd.json_normalize(results, sep="_").T
)

# Day 1 â€” Dataset Analysis & Baseline Evaluation

## 1. Dataset overview
- Total legs: 26,369
- Unique trips: 14,817
- Total features: 14
- Date range (`od_start_time`): 12 September â€“ 6 October 2018
- No missing values in the main numerical and identifier columns.
- No exact duplicate rows.

## 2. Important observations
- Actual travel time is strongly right-skewed, with some extreme values.
- `actual_time_min` does not match the duration calculated from `od_start_time` and `od_end_time`.
- Every trip has exactly one `trip_creation_time`.
- Approximately 86.45% of consecutive leg pairs within trips are connected.
- Delay ratio is a continuous target, so this is a regression problem.

## 3. Prediction target and timing
- Prediction target: `delay_ratio = actual_time_min / osrm_time_min`
- Proposed prediction moment: immediately before or at leg departure.
- Predicted excess delay minutes: `(predicted_delay_ratio - 1) Ã— osrm_time_min`.

## 4. Initial feature decisions
Potential pre-departure features:
- `source_center`
- `destination_center`
- `route_type`
- `osrm_time_min`
- `osrm_distance_km`
- Time-based features from `od_start_time`
- Potentially `trip_creation_time`-based features, subject to availability confirmation.

Excluded from the initial pre-departure feature set:
- `actual_time_min`, `delay_ratio`, `od_end_time`: outcome or post-completion information.
- `trip_uuid`: identifier, retained for grouping and splitting.
- `source_name`, `destination_name`: redundant with center IDs and contain missing values.
- `scan_rows`: timing and availability of scan events are not established.

## 5. Data splitting
Chronological, trip-level split using `trip_creation_time`:

| Partition | Trips | Legs |
|---|---:|---:|
| Training | 8,890 | 15,621 |
| Calibration | 2,963 | 5,441 |
| Test | 2,964 | 5,307 |

No trip overlap between partitions.

## 6. Baseline evaluation
Two baselines were evaluated:
1. Global median delay ratio.
2. Lane median delay ratio, with global median fallback for unseen lanes.

The lane median performed better on both calibration and test data.

- Training global median: 2.0
- Calibration lane median MAE: 0.6395
- Test lane median MAE: 0.740
- Unseen lane legs: 4.37% in calibration and 8.18% in test.

Results saved in `experiments/day1/baseline_metrics_v1.json`.

## 7. Decisions and pending confirmations
- Keep the original dataset unchanged.
- Use the lane median as the stronger baseline.
- Confirm with Person A whether planned route features and timestamped scan events are available at prediction time.
- Confirm target definition and units with Person C.

## 8. Next steps
- Build the first XGBoost regression model.
- Evaluate on calibration and test data.
- Develop Conformal Prediction intervals.
- Compare model performance against the baseline.

## Future improvements / ideas to revisit
- Explore dynamic delay prediction using timestamped scan events accumulated up to the prediction time. This requires scan-event timestamps and careful prevention of future-information leakage.

def build_features(df):
    start = pd.to_datetime(df["od_start_time"])

    X = pd.DataFrame(index=df.index)

    X["osrm_time_min"] = df["osrm_time_min"]
    X["osrm_distance_km"] = df["osrm_distance_km"]
    X["osrm_speed_kmh"] = (
        df["osrm_distance_km"] / (df["osrm_time_min"] / 60)
    )
    X["log_osrm_time"] = np.log1p(df["osrm_time_min"])

    X["start_hour"] = start.dt.hour
    X["start_dow"] = start.dt.dayofweek
    X["is_weekend"] = (start.dt.dayofweek >= 5).astype(int)

    X["is_ftl"] = (df["route_type"] == "FTL").astype(int)

    return X

from sklearn.metrics import mean_absolute_error, median_absolute_error
import numpy as np

def evaluate_model(y_true, y_pred, osrm_time):
    mae = mean_absolute_error(y_true, y_pred)
    medae = median_absolute_error(y_true, y_pred)

    excess_true = (y_true - 1) * osrm_time
    excess_pred = (y_pred - 1) * osrm_time

    excess_mae = mean_absolute_error(excess_true, excess_pred)

    return {
        "MAE_ratio": round(mae, 4),
        "MedAE_ratio": round(medae, 4),
        "MAE_excess_min": round(excess_mae, 2)
    }

def add_historical_lane_features(df, history_data=None, is_fit=False):
    result = df.copy()
    lane_cols = ["source_center", "destination_center"]

    if is_fit:
        result["lane_median_ratio"] = 1.0
        result["lane_history_count"] = 0

        for _, group in result.groupby(lane_cols, sort=False):
            group = group.sort_values("od_start_time")

            completed = group.sort_values("od_end_time")

            completed_delays = completed["delay_ratio"].to_numpy()

            completed_ends = pd.to_datetime(
              completed["od_end_time"],
              format="mixed"
             ).to_numpy(dtype="datetime64[ns]")

            pointer = 0
            past_delays = []

            for idx, row in group.iterrows():

                current_start = np.datetime64(
                    pd.Timestamp(row["od_start_time"]), "ns"
                )

                while (
                    pointer < len(completed)
                    and completed_ends[pointer] <= current_start
                ):
                    past_delays.append(completed_delays[pointer])
                    pointer += 1

                result.loc[idx, "lane_history_count"] = len(past_delays)

                if past_delays:
                    result.loc[idx, "lane_median_ratio"] = np.median(
                        past_delays
                    )

        return result

    if history_data is None:
        raise ValueError("Historical data is required.")

    lookup = history_data.set_index(lane_cols)

def add_historical_lane_features_asof(target_df, history_df):
    result = target_df.copy()

    lane_cols = ["source_center", "destination_center"]

    result["lane_median_ratio"] = 1.0
    result["lane_history_count"] = 0

    history_groups = {
        key: group.sort_values("od_end_time")
        for key, group in history_df.groupby(lane_cols, sort=False)
    }

    for key, target_group in result.groupby(lane_cols, sort=False):

        if key not in history_groups:
            continue

        hist = history_groups[key]

        end_times = hist["od_end_time"].to_numpy(
            dtype="datetime64[ns]"
        )
        delays = hist["delay_ratio"].to_numpy()

        for idx, row in target_group.iterrows():

            start_time = np.datetime64(
                row["od_start_time"], "ns"
            )

            count = np.searchsorted(
                end_times, start_time, side="right"
            )

            result.loc[idx, "lane_history_count"] = count

            if count > 0:
                result.loc[idx, "lane_median_ratio"] = np.median(
                    delays[:count]
                )

    return result

def combine_history_features(lane_df, source_df):
    result = lane_df.copy()

    use_source = result["lane_history_count"] == 0

    result["combined_history_ratio"] = np.where(
        use_source,
        source_df["source_median_ratio"],
        result["lane_median_ratio"]
    )

    result["combined_history_count"] = np.where(
        use_source,
        source_df["source_history_count"],
        result["lane_history_count"]
    )

    result["history_level"] = np.select(
        [
            result["lane_history_count"] > 0,
            source_df["source_history_count"] > 0
        ],
        ["lane", "source"],
        default="global"
    )

    return result

def add_recent_hub_delay(target_df, history_df, window_hours=48):
    hist = history_df[["source_center", "od_end_time", "delay_ratio"]].copy()
    hist["end_ts"] = pd.to_datetime(hist["od_end_time"], format="mixed")
    hist = hist.sort_values("end_ts")
    groups = {hub: (g["end_ts"].to_numpy("datetime64[ns]"), g["delay_ratio"].to_numpy())
              for hub, g in hist.groupby("source_center")}
    window = np.timedelta64(window_hours, "h")
    starts = pd.to_datetime(target_df["od_start_time"]).to_numpy("datetime64[ns]")
    hubs = target_df["source_center"].to_numpy()

    med = np.full(len(target_df), np.nan)
    cnt = np.zeros(len(target_df), dtype=int)
    for i, (hub, s) in enumerate(zip(hubs, starts)):
        if hub not in groups:
            continue
        ends, delays = groups[hub]
        hi = np.searchsorted(ends, s, side="right")           # finished before departure
        lo = np.searchsorted(ends, s - window, side="left")   # within the last 48 hours
        if hi > lo:
            med[i] = np.median(delays[lo:hi])
            cnt[i] = hi - lo
    out = target_df.copy()
    out["recent_hub_median"] = med
    out["recent_hub_count"] = cnt
    return out

def paired_trip_bootstrap(y, p_new, p_base, trips, n=2000, seed=0):

def add_prev_leg_features(d):
    out = d.copy()

def q_abs(scores, alpha):
    """Symmetric margin: covers (1-alpha) of absolute mistakes."""
    s = np.sort(scores)
    n = len(s)
    k = min(int(np.ceil((n + 1) * (1 - alpha))), n)
    return s[k - 1]

def q_signed(resid, alpha):
    """Asymmetric margins: alpha/2 allowed below, alpha/2 above."""
    s = np.sort(resid)
    n = len(s)
    k_hi = min(int(np.ceil((n + 1) * (1 - alpha / 2))), n)
    k_lo = max(int(np.floor((n + 1) * alpha / 2)), 1)
    return s[k_lo - 1], s[k_hi - 1]

def interval_metrics(y, lo, hi, osrm, alpha):
    inside = (y >= lo) & (y <= hi)
    width = hi - lo
    score = width + (2 / alpha) * np.clip(lo - y, 0, None) + (2 / alpha) * np.clip(y - hi, 0, None)
    return {
        "coverage": float(inside.mean()),
        "mean_width": float(width.mean()),
        "median_width": float(np.median(width)),
        "mean_width_min": float((width * osrm).mean()),
        "interval_score": float(score.mean())
    }

def slice_table(d, lo, hi, cols):
    d = d.assign(inside=(d["y_ratio"] >= lo) & (d["y_ratio"] <= hi), width=hi - lo)
    parts = []
    for c in cols:
        t = d.groupby(c, observed=True).agg(
            n=("inside", "size"),
            coverage=("inside", "mean"),
            mean_width=("width", "mean")
        ).reset_index()
        t.insert(0, "slice", c)
        t = t.rename(columns={c: "group"})
        parts.append(t)
    return pd.concat(parts, ignore_index=True).round(3)

# Day 3 Conclusions: Conformal Prediction Intervals

## Setup
- Frozen model: residual XGBoost (lane median + learned correction), `residual_xgb_plain_v1`.
- Calibration pile: 5,441 legs (MAE ratio 0.6288). Test pile: 5,307 legs (MAE ratio 0.7195).
- Target: delay ratio = actual time / OSRM time. Intervals are built on the ratio and converted to
  excess minutes as (ratio - 1) x OSRM time.
- Levels: 50%, 80%, 90% (alpha = 0.5, 0.2, 0.1), finite-sample corrected split conformal.

## Why asymmetric intervals
Calibration residuals (actual - predicted) are strongly right-skewed: median -0.01, 5th percentile -0.70,
95th percentile +1.88, 99th percentile +5.86. Late legs can be far later than predicted, so a longer
upper margin is needed.

## Methods compared (test pile)
| Method | Coverage 50/80/90% | Mean width, minutes 50/80/90% | Interval score 50/80/90% |
|---|---|---|---|
| A symmetric | 0.442 / 0.772 / 0.884 | 40.8 / 116.8 / 206.2 | 2.632 / 5.192 / 8.414 |
| B asymmetric | 0.445 / 0.747 / 0.873 | 43.5 / 127.0 / 224.1 | 2.631 / 5.094 / 7.914 |
| **C grouped (route type x distance tercile)** | 0.457 / 0.761 / 0.872 | **35.4 / 94.8 / 159.7** | 2.601 / 4.926 / 7.592 |
| D grouped + lane-history flag | 0.455 / 0.756 / 0.865 | 36.6 / 94.4 / 166.4 | 2.601 / 4.931 / 7.578 |
| E rolling + grouped (last 3,000 finished legs) | **0.481 / 0.792 / 0.895** | 37.1 / 105.3 / 193.2 | 2.598 / 4.920 / 7.591 |

## Main findings
1. **Static grouped conformal (C) is the saved final method.** It has the best interval score among
   the static methods and, at 90%, intervals about 30% narrower in minutes than B (160 vs 224 min) at
   similar coverage. Separate margins for FTL/Carting and short/medium/long legs equalise coverage
   across groups (0.86 to 0.88 at 90%).
2. **Coverage on the test pile is 3 to 4 points below target** for all static methods. Trip-bootstrap
   95% intervals for B exclude the targets (e.g. 90%: 0.873, interval [0.864, 0.881]).
3. **The shortfall is a calibration-age effect, not a method error.** A time-split check inside the
   calibration pile was on target (A: 0.498 / 0.810 / 0.903; B: 0.501 / 0.808 / 0.909). Test legs are

   harder (MAE 0.7195 vs 0.6288) and include more unseen lanes (8.2% vs 4.4% of legs).
   Daily coverage inside the test period is flat (about 0.86 to 0.88), pointing to a level shift
   between the calibration and test weeks rather than a trend.
4. **Rolling recalibration repairs most of the gap.** Refreshing margins with the most recent 3,000
   finished legs raised B's coverage from 0.445 / 0.747 / 0.873 to 0.484 / 0.782 / 0.893, and the
   grouped version (E) reached 0.481 / 0.792 / 0.895, at an interval score equal to C but with
   intervals about 5% / 11% / 21% wider in minutes.
5. **Lane history matters.** Unseen lanes are the weakest group (90% coverage 0.774 for B, 0.744 for C).
   Adding a low-history flag to the groups (D) raised unseen-lane coverage to 0.797 but lowered
   coverage on known lanes, with no overall interval-score gain, so C was kept.
6. The reliability curve lies slightly below the diagonal at every level (largest gap in the middle
   levels), i.e. static intervals are mildly over-confident on the test period.

## Limitations
- Data covers about three weeks (Sep to Oct 2018), one season, with a single calibration/test split.
- Legs within a trip are correlated, so effective sample sizes are smaller than leg counts.
- 2.1% to 4.6% of training legs finish after the next pile begins (mild temporal overlap, not removed).
- Rolling recalibration uses outcomes of earlier test-period legs to refresh margins for later days
  (the model itself stays frozen); this assumes outcomes become available after legs finish.
- Weather features are not used yet (weather data covers only a 7,817-leg core subset).
- Margins live on the delay-ratio scale; converting to minutes assumes the ratio scales with OSRM time.

# SupplyShield: Cascade Simulator Rules & Specification (v1)

## 1. Core Rules (Motter-Lai Style Load Redistribution)
1. **Hub Representation:** Every hub has a defined **capacity** and current **load**.
2. **Initial Disruption:** A scenario initializes by failing one or more target hubs.
3. **Load Redistribution:** A failed hub distributes its accumulated load to its **alive neighbors** in direct proportion to their **spare room** (capacity minus current load). If all neighboring hubs are at or over capacity, the load is shared in proportion to their total capacities.
4. **Failure Criterion:** A hub fails if its updated load exceeds `tolerance * capacity` (where tolerance represents the resilience threshold).
5. **Iterative Rounds:** The process repeats in discrete rounds until no new hubs fail or `max_rounds` is reached.
6. **Unserved Load:** Any load originating from a failed hub that has no remaining alive neighbors to route to is classified as **unserved load**.

---

## 2. Parameters
* **`tolerance`**: Default `1.0` (hubs fail immediately when load exceeds capacity; e.g., `1.1` allows a 10% overload buffer).
* **`max_rounds`**: Default `20` (maximum propagation steps for cascade evaluation).

---

## 3. Input & Output Formats
* **Inputs:** 
  * `capacity`: Dictionary mapping hub identifiers to numerical capacity values.
  * `load`: Dictionary mapping hub identifiers to baseline numerical load values.
  * `neighbors`: Adjacency list mapping each hub to its connected peers.
  * `initial_failed`: List of hub IDs targeted for initial failure.
* **Outputs:** 
  * `failed`: Sorted list of all hubs that failed during the cascade.
  * `n_failed`: Total count of failed hubs.
  * `n_rounds`: Total number of rounds taken for the cascade to settle.
  * `rounds`: Step-by-step audit trail of newly failed hubs and shifted loads per round.
  * `unserved_load`: Total load that could not be redistributed.
  * `unserved_share`: Ratio of unserved load to total network load.
  * `conserved`: Boolean verification flag ensuring total load equals unserved load plus alive load.

---

## 4. Modeling Disclaimer
> Capacities and baseline loads are derived or synthetic. The redistribution rule (proportional to spare room) and the tolerance are modelling assumptions, not observed behavior. Results show how fragile the network would be under these assumptions.

import numpy as np

def trigger_table(cap, load, nb, tol=1.0):
    rows = []
    for h in cap:
        spare = sum(max(tol * cap[j] - load[j], 0) for j in nb.get(h, []))

            "hub": h, 
            "degree": len(nb.get(h, [])), 
            "own_load": round(load[h], 2), 
            "neighbor_spare": round(spare, 2), 
            "load_to_spare": load[h] / spare if spare > 0 else np.inf

    return pd.DataFrame(rows).sort_values("load_to_spare", ascending=False).reset_index(drop=True)

def stress_sweep(levels, tolerances, cap, nb, max_rounds=30):
    rows = []
    for s in levels:
        load = {h: s * c for h, c in cap.items()}
        for tol in tolerances:
            res = [simulate_cascade(cap, load, nb, [h], tol, max_rounds) for h in cap]
            nf = np.array([r["n_failed"] for r in res])
            rows.append({
                "stress": round(s, 2),
                "tolerance": tol,
                "share_hubs_cascading": float((nf > 1).mean()),
                "mean_failed": float(nf.mean()),
                "max_failed": int(nf.max()),
                "mean_unserved_share": float(np.mean([r["unserved_share"] for r in res])),
                "non_converged": int(sum(not r["converged"] for r in res))
            })
    return pd.DataFrame(rows)

"""
sim/simulate.py - Cascade simulator v1 (Person B)

Load-redistribution cascade (Motter-Lai style):
- every hub has a capacity and a current load
- a failed hub's load goes to its alive neighbours, in proportion to their spare room
  (if nobody has spare room: in proportion to capacity)
- a hub fails when load > tolerance * capacity
- rounds repeat until no new hub fails
- load with nowhere to go is "unserved"
"""

from collections import defaultdict
import pandas as pd

EPS = 1e-9


def build_neighbors(edges):
    """edges: list of (hub_a, hub_b). Lanes are treated as two-way for load sharing."""
    nb = defaultdict(set)
    for a, b in edges:
        if a != b:
            nb[a].add(b)
            nb[b].add(a)
    return {k: sorted(v) for k, v in nb.items()}


def simulate_cascade(capacity, load, neighbors, initial_failed, tolerance=1.0, max_rounds=20, check_initial=True):
    cap = dict(capacity)
    cur = {n: float(load.get(n, 0.0)) for n in cap}
    bad = [h for h in initial_failed if h not in cap]
    # Guard against starting in an already-broken state
    if check_initial:
        over = [n for n in cap if cur[n] > tolerance * cap[n] + EPS]
        if over:
            raise ValueError(f"{len(over)} hubs start above tolerance x capacity, e.g., {over[:5]}")

    if bad:
        raise ValueError(f"unknown hubs in scenario: {bad}")
    
    total = sum(cur.values())
    failed_set, rounds, unserved = set(), [], 0.0
    newly = list(dict.fromkeys(initial_failed))  # drop duplicates, keep order
    
    for r in range(max_rounds + 1):
        if not newly:
            break
        failed_set.update(newly)
        add, shifted = defaultdict(float), 0.0
        
        for h in newly:
            moved, cur[h] = cur[h], 0.0
            shifted += moved
            if moved <= 0:
                continue
            
            alive = [j for j in neighbors.get(h, []) if j not in failed_set]
            if not alive:
                unserved += moved
                continue
            
            spare = {j: max(cap[j] - cur[j], 0.0) for j in alive}
            weight = spare if sum(spare.values()) > EPS else {j: cap[j] for j in alive}
            tot_w = sum(weight.values())
            
            if tot_w <= EPS:
                unserved += moved
                continue
            
            for j in alive:
                add[j] += moved * weight[j] / tot_w
                
        for j, a in add.items():
            cur[j] += a
            
        rounds.append({
            "round": r,
            "newly_failed": list(newly),
            "load_shifted": shifted,
            "total_failed": len(failed_set)
        })
        
        newly = [n for n in cap if n not in failed_set and cur[n] > tolerance * cap[n] + EPS]
        
    alive_load = {n: cur[n] for n in cap if n not in failed_set}
    util = [alive_load[n] / cap[n] for n in alive_load if cap[n] > 0]
    
    return {
        "initial_failed": list(initial_failed),
        "failed": sorted(failed_set),
        "n_failed": len(failed_set),
        "n_rounds": len(rounds),
        "rounds": rounds,
        "converged": len(newly) == 0,
        "total_load": total,
        "unserved_load": unserved,
        "unserved_share": unserved / total if total > 0 else 0.0,
        "final_load": alive_load,
        "max_utilization": max(util) if util else 0.0,
        "conserved": abs(total - (unserved + sum(alive_load.values()))) < 1e-6,
    }


def single_hub_failures(capacity, load, neighbors, tolerance=1.0, max_rounds=20, check_initial=True):
    """Fail each hub on its own and record how far the cascade spreads."""
    rows = []
    for h in capacity:
        r = simulate_cascade(capacity, load, neighbors, [h], tolerance, max_rounds, check_initial=check_initial)
        rows.append({
            "hub": h,
            "own_load": load.get(h, 0.0),
            "n_failed": r["n_failed"],
            "n_rounds": r["n_rounds"],
            "unserved_share": round(r["unserved_share"], 3),
            "max_util_alive": round(r["max_utilization"], 3)
        })
    return pd.DataFrame(rows).sort_values(["n_failed", "unserved_share"], ascending=False).reset_index(drop=True)

# SupplyShield: Cascade Simulator Rules & Specification (v1)

## 1. Core Rules (Motter-Lai Style Load Redistribution)
1. **Hub Representation:** Every hub has a defined **capacity** and current **load**.
2. **Initial Disruption:** A scenario initializes by failing one or more target hubs (`check_initial=True` safeguards against invalid starting states where load exceeds tolerance Ã— capacity).
3. **Load Redistribution:** A failed hub distributes its accumulated load to its **alive neighbors** in direct proportion to their **spare room** (capacity minus current load). If all neighboring hubs are at or over capacity, the load is shared in proportion to their total capacities.
4. **Failure Criterion:** A hub fails if its updated load exceeds `tolerance * capacity` (where tolerance represents the resilience threshold, e.g., `1.0` or `1.1`).
5. **Iterative Rounds:** The process repeats in discrete rounds until no new hubs fail or `max_rounds` is reached.
6. **Unserved Load:** Any load originating from a failed hub that has no remaining alive neighbors to route through is classified as **unserved load**.

---

## 2. Definition of a Simulation "Round"
* A **round** represents a single discrete propagation step within a simulation day where all newly overloaded hubs simultaneously shed and redistribute their excess load to their local alive neighbors. 
* The cascade terminates when a round produces zero newly failed hubs (convergence) or reaches `max_rounds`.

---

## 3. Parameters
* **`tolerance`**: Default `1.0` (hubs fail immediately when load exceeds capacity; e.g., `1.1` allows a 10% overload buffer).
* **`max_rounds`**: Default `20` (maximum propagation steps for cascade evaluation).

---

## 4. Input & Output Formats
* **Inputs:** 
  * `capacity`: Dictionary mapping hub identifiers to numerical capacity values.
  * `load`: Dictionary mapping hub identifiers to baseline numerical load values.
  * `neighbors`: Adjacency list mapping each hub to its connected peers.
  * `initial_failed`: List of hub IDs targeted for initial failure.
* **Outputs:** 
  * `failed`: Sorted list of all hubs that failed during the cascade.
  * `n_failed`: Total count of failed hubs.
  * `n_rounds`: Total number of rounds taken for the cascade to settle.
  * `rounds`: Step-by-step audit trail of newly failed hubs and shifted loads per round.
  * `unserved_load`: Total load that could not be redistributed.
  * `unserved_share`: Ratio of unserved load to total network load.
  * `conserved`: Boolean verification flag ensuring total load equals unserved load plus alive load.

---

## 5. Modeling & Validation Summary
> Capacities and baseline loads are derived from daily core network traffic (`train_legs_v1.csv`). Unit-consistent validation (`single_failure_real_loads_v3.csv`) confirms that under real daily load profiles, the network is robust, resulting in **0 multi-hub cascades** once capacity-load boundary artifacts are reconciled.

# Safe-Plan & Risk Analysis (Hub D Capacity Uncertainty)

## Decision Rule: Express Van Utilization
The optimizer routes flow through the express van only when the express van cost per parcel is strictly less than the penalty for an undelivered parcel (20).

## Safe-Plan Paragraph (CVaR Context)
A baseline planning model that assumes Hub D can consistently handle 60 parcels will route up to 60 units through it to minimize daily operating costs. However, on days when Hub D's capacity drops to 30 or 0, this aggressive baseline causes severe capacity overflows and massive penalty spikes.

Conditional Value at Risk (CVaR) is not the single absolute worst-case scenario; rather, it represents the expected average cost across the worst slice of outcomes (e.g., the worst 20% of days). A CVaR-optimized plan proactively diverts volume to slightly higher-cost secondary paths, effectively removing catastrophic cost spikes on bad days while accepting a small increase in standard operating costs.

## 2. Evaluation Findings

* **Deterministic LP vs. Greedy Baseline:** The deterministic LP assumes a fixed, ideal capacity for Hub D at all times without accounting for potential capacity drops or risk. Because it plans under the assumption that Hub D will never fail, it heavily over-allocates traffic to Hub D. When Hub D suffers capacity bottlenecks during bad days, the deterministic LP incurs extreme spillover penalties (leading to a worst-day cost of 1700), which severely drags down its overall average cost compared to the greedy baseline.

* **CVaR Plan Behavior:** The CVaR LP explicitly optimizes against worst-case tail outcomes. Instead of routing traffic through cheaper but volatile routes (such as Hub D), it proactively routes demand through consistently available, risk-averse alternative paths (costing 6 per unit instead of 5). By completely bypassing the volatile bottleneck, its daily operational cost remains flat at 810 regardless of Hub D's daily capacity fluctuations.

* **Expected-Cost LP Significance:** While the expected-cost LP shows a slightly lower average cost (681 vs. 684â€”a 3-point difference), this gap is **not statistically meaningful**. Given that the seed-to-seed standard deviation across runs is approximately 10, the 3-point difference falls well within normal sampling noise and variance.

* **Management Recommendations:**
  * **Risk-Averse Manager (Fears Bad Days):** Recommend **Plan 5 (CVaR LP)** or **Plan 4 (Expected-cost LP)**. Plan 5 guarantees a strict upper bound of 810 with zero daily variance, while Plan 4 caps worst-day exposure at 1220 (compared to 1500+ for greedy/deterministic).
  * **Cost-Focused Manager (Cares About Average):** Recommend **Plan 4 (Expected-cost LP)**. It achieves the lowest overall average cost (~681) while providing significantly better protection against worst-case spikes than the greedy or deterministic baselines.

> **Note on Model Scope:** These results are derived from a simplified toy network with fixed parameters (spillover penalty of 20, specific failure probabilities, and alternative route costs of 6 vs. 5). These numbers illustrate policy behavior and risk-tradeoff mechanics rather than exact empirical predictions for real-world logistics.

* **Network-Wide Re-optimization:** Allowing background traffic to adjust routes (LP) reduces closure-induced extra costs from 28.4 to 25.2 leg-hours (~11% savings over Greedy).

# Training-only rain comparison

Dry means exactly 0 mm of source departure-day rainfall.
Rainy thresholds are 1, 5 and 10 mm; intermediate amounts are excluded.
Each compared lane needs at least three legs per condition.
The additional multi-day check requires at least two dates per condition.

Lane effects are rainy median travel-time ratio minus dry median ratio.
The reported overall median gives each eligible lane equal weight.
Different thresholds can include different lanes, so their results are
not a dose-response estimate.

This is descriptive analysis, not a causal or statistical significance test.
Trips sharing dates, locations and weather grids are not independent.
City coordinates and daily endpoint rainfall are approximate exposures.
Same-day historical rainfall is not information available at departure.
The short training period limits generalisation.

## Results

```text
 rain_threshold_mm  eligible_legs  dry_legs  rainy_legs  comparable_lanes  represented_legs  median_difference  positive_lane_pct  multi_day_lanes  multi_day_median_difference
               1.0           5118      1479        2156               113              2179            -0.0058            47.7876              110                      -0.0056
               5.0           5118      1479         906                81              1471            -0.0048            49.3827               79                       0.0221
              10.0           5118      1479         558                54              1134             0.0594            62.9630               44                       0.0468
```
### Rain comparison finding

Training-only within-lane comparisons showed near-zero median differences
at rainfall thresholds of 1 and 5 mm. At 10 mm, the median travel-time-ratio
difference was +0.0594 across 54 lanes, with 62.96% showing an increase.
Requiring at least two dates per condition retained 44 lanes and a positive
median difference of +0.0468.

These descriptive results suggest a possible association under heavier
rainfall, but do not establish causation or statistical significance.
The lane sample changes with the rainfall threshold. Weather-based
capacity reductions or delay penalties have not been calibrated from
these results.

"""
SupplyShield - Person C, Day 8-9: ALL traffic shares the hubs.

c_real_graph.py rerouted ONE lane's traffic through empty hubs (optimistic). Here every lane of the
core graph is a flow of its own ('commodity': origin, destination, legs per day). In normal times each
flow uses its own direct lane. Then ONE lane closes and we ask three plans to cope:
   1. do nothing   : the closed lane's legs are lost, every other flow keeps its direct lane
   2. greedy       : the closed lane's flow takes the cheapest detours, using only the room the
                     other (unchanged) flows leave free at each hub
   3. LP           : every flow may change route; minimises total travel hours + penalty for lost legs
All flows share the hub capacity. Capacity of a hub = A's capacity_est, which is defined on its
OUTGOING legs per day, so the rule is: all flows leaving a hub together <= capacity_est.

Run from the project root:  python opt/c_multiflow.py        (needs opt/c_real_graph.py, c_toy_recovery.py)
Output: experiments/c_multiflow_closures.csv
(extra_cost = how much worse the whole network gets because of the closure, under that plan)

ASSUMPTIONS (write them in docs/c_notes.md):
  - demand of a flow = trips / 14 legs per day (training days)
  - traffic that leaves the core is ignored, but capacity_est counts it, so capacities are still a bit optimistic
  - cost = travel hours; a lost leg costs PENALTY_HOURS; each closure is tested alone
"""
import time
from pathlib import Path

import numpy as np
import pandas as pd
from scipy.optimize import linprog
from scipy.sparse import coo_matrix

import c_real_graph as rg
import c_toy_recovery as c

BIG = 10 ** 9


def build_model(G):
    nodes = list(G.nodes)
    nidx = {n: i for i, n in enumerate(nodes)}
    edges = list(G.edges)
    eidx = {e: i for i, e in enumerate(edges)}
    N, E = len(nodes), len(edges)
    src = np.array([nidx[u] for u, _ in edges])
    dst = np.array([nidx[v] for _, v in edges])
    cost = np.array([G.edges[e]["cost"] for e in edges])
    flows = [(e[0], e[1], G.edges[e]["trips"] / rg.TRAIN_DAYS) for e in edges]   # (origin, destination, q)
    return nodes, nidx, edges, eidx, N, E, src, dst, cost, flows


def solve_multiflow(G, closed=None):
    """Minimum total cost over all flows sharing hub capacity. `closed` = one lane (u, v) that is shut."""
    nodes, nidx, edges, eidx, N, E, src, dst, cost, flows = build_model(G)
    K = len(flows)
    nvar = K * E + K                                   # x[k, e] and unmet[k]
    ks = np.repeat(np.arange(K), E)
    es = np.tile(np.arange(E), K)
    col = ks * E + es

    # flow balance: for flow k at node n:  out - in + [n=origin]*unmet - [n=dest]*unmet = [n=origin]*q - [n=dest]*q
    rows = np.concatenate([ks * N + src[es], ks * N + dst[es]])
    cols = np.concatenate([col, col])
    vals = np.concatenate([np.ones(K * E), -np.ones(K * E)])
    o_idx = np.array([nidx[o] for o, _, _ in flows])
    d_idx = np.array([nidx[d] for _, d, _ in flows])
    q = np.array([qq for _, _, qq in flows])
    karr = np.arange(K)
    rows = np.concatenate([rows, karr * N + o_idx, karr * N + d_idx])
    cols = np.concatenate([cols, K * E + karr, K * E + karr])
    vals = np.concatenate([vals, np.ones(K), -np.ones(K)])
    A_eq = coo_matrix((vals, (rows, cols)), shape=(K * N, nvar)).tocsr()
    b_eq = np.zeros(K * N)
    b_eq[karr * N + o_idx] += q
    b_eq[karr * N + d_idx] -= q

    # hub capacity: everything leaving hub n (all flows together) <= capacity_est(n)
    A_ub = coo_matrix((np.ones(K * E), (src[es], col)), shape=(N, nvar)).tocsr()
    b_ub = np.array([G.nodes[n]["cap"] for n in nodes])

    bounds_hi = np.full(nvar, np.inf)
    bounds_hi[K * E:] = q                              # cannot lose more than the demand
    if closed is not None:
        bounds_hi[np.arange(K) * E + eidx[closed]] = 0.0
    objective = np.concatenate([np.tile(cost, K), np.full(K, rg.PENALTY_HOURS)])
    res = linprog(objective, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=np.column_stack([np.zeros(nvar), bounds_hi]), method="highs")
    if not res.success:
        raise RuntimeError(res.message)
    unmet = float(res.x[K * E:].sum())
    travel = float(np.dot(np.tile(cost, K), res.x[:K * E]))
    return travel + rg.PENALTY_HOURS * unmet, unmet


def baseline_plans(G, lane):
    """do nothing and greedy for the closure of `lane`; every other flow stays on its direct lane."""
    u, v = lane
    q = G.edges[lane]["trips"] / rg.TRAIN_DAYS
    normal_cost = sum(G.edges[e]["cost"] * G.edges[e]["trips"] / rg.TRAIN_DAYS for e in G.edges)
    other_cost = normal_cost - q * G.edges[lane]["cost"]

    # room left at each hub once all OTHER flows sit on their direct lanes
    load = {n: 0.0 for n in G.nodes}
    for (a, b) in G.edges:
        if (a, b) != lane:
            load[a] += G.edges[(a, b)]["trips"] / rg.TRAIN_DAYS
    residual = {n: max(0.0, G.nodes[n]["cap"] - load[n]) for n in G.nodes}

    H = G.copy()
    for n in H.nodes:
        H.nodes[n]["cap"] = residual[n]
    for e in H.edges:
        H.edges[e]["cap"] = BIG
    H.edges[lane]["cap"] = 0.0
    H.nodes[u]["cap"] = H.nodes[v]["cap"] = BIG
    c.SRC, c.DST, c.DEMAND, c.PENALTY = u, v, q, rg.PENALTY_HOURS
    flows, delivered = c.greedy(H)
    detour = sum(H.edges[e]["cost"] * f for e, f in flows.items())
    unmet = q - delivered
    return {"do nothing": (other_cost + rg.PENALTY_HOURS * q, q),
            "greedy": (other_cost + detour + rg.PENALTY_HOURS * unmet, unmet)}, normal_cost

"""
SupplyShield - Person C, Day 6-7: the toy optimizer on the REAL core graph.

What it does
  1. Loads Person A's core_nodes.csv and core_edges.csv into a directed graph.
       node 'cap'  = hub capacity in truck legs per day   (A's capacity_est)
       edge 'cost' = median travel time of the lane in hours
       edge 'cap'  = unlimited (hub capacities are what limit the flow)
  2. Takes the 10 busiest lanes. For each one, pretend the lane is CLOSED (flood, strike) and
     reroute that lane's daily traffic through other hubs. Compare three plans:
       do nothing (the traffic is lost)  |  greedy  |  optimizer (LP)
  3. Saves experiments/c_real_lane_closures.csv and prints the run time of the optimizer.

Run from the project root:
    python opt/c_real_graph.py            # hubs are assumed EMPTY apart from the rerouted flow (optimistic)
    python opt/c_real_graph.py --spare    # hubs keep only their SPARE capacity = capacity_est - normal daily traffic (pessimistic)
Needs: opt/c_toy_recovery.py (it reuses its solver), pandas numpy scipy networkx.

ASSUMPTIONS (write them in docs/c_notes.md):
  - cost unit = hours of travel; penalty for a leg that is never delivered = PENALTY_HOURS (a guess; test 24/48/96)
  - each closure is tested ALONE: other traffic in the network is ignored, so capacities are optimistic.
    The multi-commodity version (several flows sharing hub capacity) is the next step.
"""
import sys
import time

import networkx as nx
import numpy as np
import pandas as pd

import c_toy_recovery as c

NODES_CSV = "data/processed/core_nodes.csv"
EDGES_CSV = "data/processed/core_edges.csv"
# used only if core_edges.csv has no travel-time column (Person A's file currently does not)
LEGS_CANDIDATES = ["data/processed/core_train_legs.csv", "data/processed/legs.csv"]
TRAIN_CUTOFF = "2018-09-26"   # same training cut as Person A's core and Person B's split
TIME_DIVISOR = 60.0        # Delhivery times are in minutes -> hours
TRAIN_DAYS = 14            # days the 'trips' column was counted over
PENALTY_HOURS = 96.0       # cost of one undelivered leg, in hours of travel
BIG = 10 ** 9


def find(df, candidates, what, substrings=(), required=True):
    for name in candidates:
        if name in df.columns:
            return name
    for col in df.columns:
        if substrings and all(sub in col.lower() for sub in substrings):
            return col
    if required:
        raise SystemExit(f"Could not find the {what} column. Columns are: {df.columns.tolist()}")
    return None


def lane_times_from_legs():
    """Median travel time (minutes) of every lane, computed from the leg table, training days only."""
    path = next((p for p in LEGS_CANDIDATES if Path(p).exists()), None)
    if path is None:
        raise SystemExit("core_edges.csv has no travel-time column and no legs file was found in "
                         f"{LEGS_CANDIDATES}. Ask Person A to add median_actual_time to core_edges.csv.")
    legs = pd.read_csv(path)
    l_src = find(legs, ["source_center"], "legs source")
    l_dst = find(legs, ["destination_center"], "legs destination")
    l_time = find(legs, ["actual_time"], "actual travel time", ("actual", "time"))
    if "core_train_legs" not in path and "od_start_time" in legs.columns:
        legs = legs[pd.to_datetime(legs["od_start_time"], errors="coerce") < pd.Timestamp(TRAIN_CUTOFF)]
    print(f"lane travel times computed from {path} (column '{l_time}', {len(legs):,} training legs)")
    med = legs.groupby([l_src, l_dst])[l_time].median().rename("median_time").reset_index()
    return med.rename(columns={l_src: "src_key", l_dst: "dst_key"})


def mean_daily_outgoing():
    """Normal traffic: average legs per day leaving each hub during the training days."""
    path = next((p for p in LEGS_CANDIDATES if Path(p).exists()), None)
    if path is None:
        raise SystemExit("--spare needs a legs file (core_train_legs.csv or legs.csv) in data/processed")
    legs = pd.read_csv(path)
    src = find(legs, ["source_center"], "legs source")
    if "core_train_legs" not in path and "od_start_time" in legs.columns:
        legs = legs[pd.to_datetime(legs["od_start_time"], errors="coerce") < pd.Timestamp(TRAIN_CUTOFF)]
    return legs[src].value_counts() / TRAIN_DAYS


def load_core_graph(nodes_csv=NODES_CSV, edges_csv=EDGES_CSV, spare=False):
    nodes, edges = pd.read_csv(nodes_csv), pd.read_csv(edges_csv)
    n_id = find(nodes, ["hub", "hub_id", "node", "node_id", "center"], "hub id")
    n_cap = find(nodes, ["capacity_est", "capacity", "cap"], "capacity", ("cap",))
    e_src = find(edges, ["source", "source_center", "origin", "from"], "lane source")
    e_dst = find(edges, ["destination", "destination_center", "dest", "to"], "lane destination")
    e_trips = find(edges, ["trips", "trip_count"], "trips", ("trip",))
    e_time = find(edges, ["median_actual_time", "median_time", "median_travel_time"],
                  "median travel time", ("median", "time"), required=False)
    if e_time is None:                                   # fall back to the leg table
        med = lane_times_from_legs()
        edges = edges.merge(med, left_on=[e_src, e_dst], right_on=["src_key", "dst_key"], how="left")
        e_time = "median_time"
        lost = int(edges[e_time].isna().sum())
        if lost:
            print(f"WARNING: {lost} of {len(edges)} lanes have no travel time and are dropped")
        edges = edges.dropna(subset=[e_time])
    print(f"columns used -> hub:{n_id} cap:{n_cap} | source:{e_src} dest:{e_dst} trips:{e_trips} time:{e_time}")

    cap = pd.to_numeric(nodes[n_cap], errors="coerce")
    missing = int(cap.isna().sum())
    if missing:
        print(f"WARNING: {missing} hubs have no capacity; using the median ({cap.median():.0f})")
        cap = cap.fillna(cap.median())
    if spare:
        normal = nodes[n_id].map(mean_daily_outgoing()).fillna(0.0)
        cap = (cap - normal).clip(lower=0.0)
        print(f"SPARE mode: capacity = capacity_est - normal daily traffic; hubs with zero spare: {int((cap < 0.5).sum())}")
    G = nx.DiGraph()
    for hub, k in zip(nodes[n_id], cap):
        G.add_node(hub, cap=float(k))
    for r in edges.itertuples(index=False):
        s, d = getattr(r, e_src), getattr(r, e_dst)
        if s in G and d in G and s != d:
            G.add_edge(s, d, cost=float(getattr(r, e_time)) / TIME_DIVISOR,
                       cap=BIG, trips=float(getattr(r, e_trips)))
    hours = pd.Series([d["cost"] for _, _, d in G.edges(data=True)])
    print("lane travel time in HOURS (sanity check, expect a few hours):", hours.describe().round(2).to_dict())
    return G


def do_nothing_od(G_before, G_after):
    """Keep using the original cheapest path; anything that cannot pass on it is lost."""
    path = nx.shortest_path(G_before, c.SRC, c.DST, weight="cost")
    limits = [c.DEMAND]
    limits += [G_after.nodes[n]["cap"] for n in path[1:-1]]
    limits += [G_after.edges[e]["cap"] for e in zip(path, path[1:])]
    sent = max(0.0, min(limits))
    return {e: sent for e in zip(path, path[1:])}, sent


def run_closure(G, lane, demand):
    u, v = lane
    c.SRC, c.DST, c.DEMAND, c.PENALTY = u, v, demand, PENALTY_HOURS
    before = G.copy()
    before.nodes[u]["cap"] = before.nodes[v]["cap"] = BIG       # the two ends are not capacity-limited
    after = before.copy()
    after.edges[lane]["cap"] = 0.0                               # the lane is closed
    out = {}
    for name, fn in [("do nothing", lambda: do_nothing_od(before, after)),
                     ("greedy", lambda: c.greedy(after)),
                     ("optimizer", lambda: c.solve_lp(after))]:
        t0 = time.perf_counter()
        flows, delivered = fn()
        secs = time.perf_counter() - t0
        res = c.evaluate(after, flows, delivered)
        out[name] = {**res, "seconds": round(secs, 3)}
    base = demand * G.edges[lane]["cost"]
    return base, out

    print("\nLegs never delivered (unmet), summed over the 10 closures:\n",

    print("\nSlowest optimizer solve (seconds):", table[table["plan"] == "optimizer"]["seconds"].max())

========== FILE: opt\c_cvar_toy.py ==========
"""
SupplyShield - Person C, step 2: a plan chosen BEFORE we know how bad tomorrow is.
Hub D usually handles 60 parcels a day, but some days only 30 (or 0). We must decide today how many
parcels to send through each route. Parcels sent to D beyond its real capacity that day are lost
(each costs PENALTY).

Two ways to choose the plan:
  expected-cost plan : cheapest on an AVERAGE day
  CVaR plan          : cheapest on the WORST days (CVaR = average cost of the worst (1 - ALPHA) share of days)

Run from the project root:  python opt/c_cvar_toy.py      (needs scipy numpy networkx pandas)
Put this file next to c_toy_recovery.py in opt/.
"""
from pathlib import Path

import numpy as np
import pandas as pd
from scipy.optimize import linprog

import c_toy_recovery as c

HUB = "D"                                   # the hub whose capacity is uncertain
SCENARIOS = [(0.7, 60), (0.2, 30), (0.1, 0)]  # (probability, capacity of hub D that day)
ALPHA = 0.8                                 # CVaR looks at the worst 20% of days


def build_graph():
    G = c.build_toy_graph()
    # make the safe route (via E and F) pricier, so playing safe really costs something
    G.edges[("E", "F")]["cost"] = 6
    return G


def solve(G, lam):
    """lam = 0 -> minimise the average cost; lam = 1 -> minimise CVaR; in between = a mix."""
    edges = list(G.edges)
    n_e, n_s = len(edges), len(SCENARIOS)
    idx = {e: i for i, e in enumerate(edges)}
    # variables: [edge flows ..., overflow_s ..., z_s ..., t]
    n = n_e + 2 * n_s + 1
    o0, z0, t = n_e, n_e + n_s, n - 1
    p = np.array([s[0] for s in SCENARIOS])
    ship = np.array([G.edges[e]["cost"] for e in edges])

    cost = np.zeros(n)
    cost[:n_e] = (1 - lam) * ship                          # shipping cost, averaged
    cost[o0:o0 + n_s] = (1 - lam) * c.PENALTY * p          # expected loss from overflow
    cost[z0:z0 + n_s] = lam * p / (1 - ALPHA)              # CVaR part
    cost[t] = lam
    # (for lam > 0 the CVaR term also needs the shipping cost: it sits inside z_s via the constraint below)

    A_ub, b_ub, A_eq, b_eq = [], [], [], []
    row = np.zeros(n)
    for e in G.out_edges(c.SRC):
        row[idx[e]] = 1
    A_eq.append(row); b_eq.append(c.DEMAND)                # every parcel gets a route

    for node in G.nodes:
        if node in (c.SRC, c.DST):
            continue
        row = np.zeros(n)
        for e in G.in_edges(node):
            row[idx[e]] += 1
        for e in G.out_edges(node):
            row[idx[e]] -= 1
        A_eq.append(row); b_eq.append(0)
        row = np.zeros(n)
        for e in G.in_edges(node):
            row[idx[e]] = 1
        cap = max(cap_s for _, cap_s in SCENARIOS) if node == HUB else G.nodes[node]["cap"]
        A_ub.append(row); b_ub.append(cap)

    for s, (_, cap_s) in enumerate(SCENARIOS):
        row = np.zeros(n)                                  # overflow_s >= inflow_D - cap_s
        for e in G.in_edges(HUB):
            row[idx[e]] = 1
        row[o0 + s] = -1
        A_ub.append(row); b_ub.append(cap_s)
        row = np.zeros(n)                                  # z_s >= scenario_cost_s - t
        row[:n_e] = ship
        row[o0 + s] = c.PENALTY
        row[z0 + s] = -1
        row[t] = -1
        A_ub.append(row); b_ub.append(0)

    bounds = [(0, G.edges[e]["cap"]) for e in edges] + [(0, None)] * (2 * n_s) + [(None, None)]
    res = linprog(cost, A_ub=np.array(A_ub), b_ub=b_ub, A_eq=np.array(A_eq), b_eq=b_eq,
                  bounds=bounds, method="highs")
    if not res.success:
        raise RuntimeError(res.message)
    return {e: float(res.x[idx[e]]) for e in edges}


def report(G, flows):
    ship = sum(G.edges[e]["cost"] * f for e, f in flows.items())
    into_d = sum(f for (u, v), f in flows.items() if v == HUB)
    day_costs = [ship + c.PENALTY * max(0.0, into_d - cap_s) for _, cap_s in SCENARIOS]
    p = np.array([s[0] for s in SCENARIOS])
    order = np.argsort(day_costs)[::-1]                    # worst days first
    left, tail = 1 - ALPHA, 0.0
    for i in order:                                        # average of the worst (1-ALPHA) share
        take = min(p[i], left)
        tail += take * day_costs[i]
        left -= take
        if left <= 1e-12:
            break
    return {"parcels via D": round(into_d, 1), "shipping": round(ship, 1),
            **{f"cost if D={cap}": round(d, 1) for (_, cap), d in zip(SCENARIOS, day_costs)},
            "average cost": round(float(np.dot(p, day_costs)), 1),
            f"CVaR {ALPHA:.0%}": round(tail / (1 - ALPHA), 1)}


if __name__ == "__main__":
    G = build_graph()
    rows = []
    for name, lam in [("expected-cost plan", 0.0), ("mixed plan", 0.5), ("CVaR plan", 1.0)]:
        rows.append({"plan": name, **report(G, solve(G, lam))})
    table = pd.DataFrame(rows)
    pd.set_option("display.width", 200)
    print(table.to_string(index=False))
    Path("experiments").mkdir(exist_ok=True)
    table.to_csv("experiments/c_cvar_toy.csv", index=False)
    print("\nSaved experiments/c_cvar_toy.csv")

"""
SupplyShield - Person C, Day 4-5: compare 5 plans over many sampled 'tomorrows'.

Every plan is decided TODAY, before we know how much hub D can handle tomorrow.
Plans 1-3 assume D is fine (capacity 60). Plans 4-5 know the odds of a bad day.
  1. do nothing        : keep using the single cheapest original road
  2. greedy            : fill the cheapest open roads first
  3. deterministic LP  : best split of parcels assuming D = 60
  4. expected-cost LP  : best on an average day (knows the odds)
  5. CVaR LP           : best on the worst days (knows the odds)

Then we sample 1000 tomorrows (D handles 60 / 30 / 0 with the probabilities in SCENARIOS) and
measure each plan. Cost of a day = shipping + PENALTY for each parcel never planned
+ PENALTY for each parcel sent to D beyond what D can handle that day.
CVaR80 = average cost of the worst 20% of days.

Run from the project root:  python opt/c_policy_comparison.py
Needs c_toy_recovery.py and c_cvar_toy.py in the same folder (opt/).
"""
from pathlib import Path

import numpy as np
import pandas as pd

import c_cvar_toy as cv
import c_toy_recovery as c

N_DAYS = 1000
SEEDS = [0, 1, 2, 3, 4]
ALPHA = 0.8


def plans():
    G = cv.build_graph()
    out = {}
    out["1 do nothing"] = c.do_nothing(G, G)             # (flows, delivered)
    out["2 greedy"] = c.greedy(G)
    out["3 deterministic LP"] = c.solve_lp(G)
    out["4 expected-cost LP"] = (cv.solve(G, lam=0.0), c.DEMAND)
    out["5 CVaR LP"] = (cv.solve(G, lam=1.0), c.DEMAND)
    return G, out


def day_costs(G, flows, delivered, caps_d):
    ship = sum(G.edges[e]["cost"] * f for e, f in flows.items())
    into_d = sum(f for (u, v), f in flows.items() if v == cv.HUB)
    never_planned = c.PENALTY * (c.DEMAND - delivered)
    return ship + never_planned + c.PENALTY * np.maximum(0.0, into_d - caps_d)


def cvar(costs, alpha=ALPHA):
    k = int(round(len(costs) * (1 - alpha)))
    return float(np.sort(costs)[::-1][:k].mean())


if __name__ == "__main__":
    G, all_plans = plans()
    probs = [p for p, _ in cv.SCENARIOS]
    levels = [cap for _, cap in cv.SCENARIOS]
    rows = []
    for seed in SEEDS:
        caps_d = np.random.default_rng(seed).choice(levels, size=N_DAYS, p=probs)
        for name, (flows, delivered) in all_plans.items():
            d = day_costs(G, flows, delivered, caps_d)
            rows.append({"policy": name, "seed": seed, "mean_cost": d.mean(),
                         "cvar80": cvar(d), "worst_day": d.max()})
    res = pd.DataFrame(rows)
    summary = res.groupby("policy")[["mean_cost", "cvar80", "worst_day"]].agg(["mean", "std"]).round(1)
    pd.set_option("display.width", 200)
    print(summary.to_string())
    Path("experiments").mkdir(exist_ok=True)
    res.to_csv("experiments/c_toy_policy_comparison.csv", index=False)
    print("\nSaved experiments/c_toy_policy_comparison.csv")

import os
import pandas as pd
import c_toy_recovery as c

def safe_eval(G, flows, delivered):
    res = c.evaluate(G, flows, delivered)
    return (res[0], res[1]) if isinstance(res, tuple) else (res.get("total_cost", res), res.get("unmet", 0))

def run_scenarios():
    os.makedirs("experiments", exist_ok=True)
    rows = []

    # 1. Single Hub Failures (Full & Partial) across Penalty Costs
    for penalty in [8, 20, 40]:
        c.PENALTY = penalty
        for node in ["A", "B", "C", "D", "E", "F"]:
            for remaining in [0.0, 0.5]:
                G0 = c.build_toy_graph()
                G1 = c.fail_node(G0, node, remaining)

                for name, func in [("Do Nothing", lambda: c.do_nothing(G0, G1)), 
                                   ("Greedy", lambda: c.greedy(G1)), 
                                   ("Optimizer", lambda: c.solve_lp(G1))]:
                    fl, deliv = func()
                    cost, unmet = safe_eval(G1, fl, deliv)
                    rows.append({
                        "scenario_type": "single_hub",
                        "penalty": penalty,
                        "failed_hubs": f"{node} ({int((1-remaining)*100)}% down)",
                        "van_cost": 12,
                        "plan": name,
                        "total_cost": cost,
                        "unmet": unmet
                    })

    # 2. Double Hub Failures
    c.PENALTY = 20
    for n1, n2 in [("A", "E"), ("B", "D"), ("C", "F"), ("A", "C")]:
        G0 = c.build_toy_graph()
        G1 = c.fail_node(c.fail_node(G0, n1, 0.0), n2, 0.0)

        for name, func in [("Do Nothing", lambda: c.do_nothing(G0, G1)), 
                           ("Greedy", lambda: c.greedy(G1)), 
                           ("Optimizer", lambda: c.solve_lp(G1))]:
            fl, deliv = func()
            cost, unmet = safe_eval(G1, fl, deliv)
            rows.append({
                "scenario_type": "double_hub",
                "penalty": 20,
                "failed_hubs": f"{n1}+{n2} (100% down)",
                "van_cost": 12,
                "plan": name,
                "total_cost": cost,
                "unmet": unmet
            })

    # 3. Express Van Cost Variations (Tested on Hub D Failure)
    c.PENALTY = 20
    for van_cost in [12, 18, 19, 21, 25]:
        G0 = c.build_toy_graph()
        G0.edges[("S", "T")]["cost"] = van_cost
        G1 = c.fail_node(G0, "D", 0.0)

        for name, func in [("Do Nothing", lambda: c.do_nothing(G0, G1)), 
                           ("Greedy", lambda: c.greedy(G1)), 
                           ("Optimizer", lambda: c.solve_lp(G1))]:
            fl, deliv = func()
            cost, unmet = safe_eval(G1, fl, deliv)
            express_used = fl.get(("S", "T"), 0)
            rows.append({
                "scenario_type": "van_cost_sensitivity",
                "penalty": 20,
                "failed_hubs": f"D (van_cost={van_cost})",
                "van_cost": van_cost,
                "plan": name,
                "total_cost": cost,
                "unmet": unmet,
                "express_flow": express_used
            })

    df = pd.DataFrame(rows)
    output_path = "experiments/c_toy_scenarios.csv"
    df.to_csv(output_path, index=False)
    print(f"Successfully saved {len(df)} scenarios to '{output_path}'!\n")

    # --- PRINT SUMMARY TABLES ---
    print("=== AVERAGE TOTAL COST PER PLAN ===")
    avg_costs = df[df["scenario_type"] != "van_cost_sensitivity"].groupby("plan")["total_cost"].mean()
    print(avg_costs.round(2).to_string())
    
    print("\n=== EXPRESS VAN SENSITIVITY (Hub D Failure - Optimizer) ===")
    express_df = df[(df["scenario_type"] == "van_cost_sensitivity") & (df["plan"] == "Optimizer")][["van_cost", "express_flow", "total_cost"]]
    print(express_df.to_string(index=False))

"""
SupplyShield - Person C starter (Day 1 + Day 2).
A toy 8-hub network, a hub failure, and three ways to respond:
  1. do nothing     - keep sending along the original cheapest route
  2. greedy         - cheapest surviving route first, then the next, until nothing is left
  3. optimizer (LP) - best split of parcels over all routes (minimum total cost)
Run from the project root:  python opt/c_toy_recovery.py   (needs: pip install scipy numpy networkx pandas)
Put this file in opt/ and import the functions from your notebook later.
"""
from pathlib import Path

import networkx as nx
import numpy as np
import pandas as pd
from scipy.optimize import linprog

SRC, DST = "S", "T"
DEMAND = 100          # parcels per day that must go from S to T
PENALTY = 20.0        # cost of one parcel that is NOT delivered (angry customer, refund)


def build_toy_graph():
    """Nodes have a daily capacity; edges have a cost per parcel (and optionally a cap)."""
    G = nx.DiGraph()
    caps = {"A": 60, "B": 60, "C": 50, "D": 60, "E": 40, "F": 40}
    for n, c in caps.items():
        G.add_node(n, cap=c)
    G.add_node(SRC, cap=10**6)
    G.add_node(DST, cap=10**6)
    edges = [("S", "A", 1), ("S", "B", 2), ("A", "C", 2), ("A", "D", 3), ("B", "D", 1),
             ("B", "E", 2), ("C", "T", 2), ("D", "T", 2), ("E", "F", 1), ("F", "T", 1)]
    for u, v, cost in edges:
        G.add_edge(u, v, cost=cost, cap=10**6)
    G.add_edge("S", "T", cost=12, cap=30)   # express van: expensive, limited
    return G


def fail_node(G, node, remaining=0.0):
    """Return a copy where `node` keeps only `remaining` (0 = fully down) of its capacity."""
    H = G.copy()
    H.nodes[node]["cap"] = H.nodes[node]["cap"] * remaining
    return H


def evaluate(G, flows, delivered):
    """Total cost = shipping cost of the flows + penalty for undelivered parcels."""
    ship = sum(G.edges[e]["cost"] * f for e, f in flows.items())
    unmet = DEMAND - delivered
    return {"shipping": round(ship, 1), "unmet": round(unmet, 1),
            "total_cost": round(ship + PENALTY * unmet, 1)}


# ---------------------------------------------------------------- optimizer
def solve_lp(G):
    """Minimum-cost split of DEMAND parcels over all routes, respecting node/edge capacities.
    Variables: one flow per edge, plus 'unmet' (parcels we fail to deliver)."""
    edges = list(G.edges)
    n_e = len(edges)                      # variable i < n_e is the flow on edges[i]; last one is unmet
    idx = {e: i for i, e in enumerate(edges)}
    cost = np.array([G.edges[e]["cost"] for e in edges] + [PENALTY])

    A_eq, b_eq, A_ub, b_ub = [], [], [], []

    row = np.zeros(n_e + 1)               # parcels leaving the source = demand - unmet
    for e in G.out_edges(SRC):
        row[idx[e]] = 1
    row[-1] = 1
    A_eq.append(row)
    b_eq.append(DEMAND)

    for n in G.nodes:
        if n in (SRC, DST):
            continue
        row = np.zeros(n_e + 1)           # parcels do not vanish at a hub: in = out
        for e in G.in_edges(n):
            row[idx[e]] += 1
        for e in G.out_edges(n):
            row[idx[e]] -= 1
        A_eq.append(row)
        b_eq.append(0)

        row = np.zeros(n_e + 1)           # a hub can only handle so many per day
        for e in G.in_edges(n):
            row[idx[e]] = 1
        A_ub.append(row)
        b_ub.append(G.nodes[n]["cap"])

    bounds = [(0, G.edges[e]["cap"]) for e in edges] + [(0, DEMAND)]
    res = linprog(cost, A_ub=np.array(A_ub), b_ub=b_ub, A_eq=np.array(A_eq), b_eq=b_eq,
                  bounds=bounds, method="highs")
    if not res.success:
        raise RuntimeError(res.message)
    flows = {e: float(res.x[idx[e]]) for e in edges}
    return flows, DEMAND - float(res.x[-1])


# ---------------------------------------------------------------- baselines
def do_nothing(G_before, G_after):
    """Keep using the cheapest ORIGINAL route. If it goes through a dead hub, parcels are lost."""
    path = nx.shortest_path(G_before, SRC, DST, weight="cost")
    cap = min(G_after.nodes[n]["cap"] for n in path[1:-1])
    sent = min(DEMAND, cap)
    flows = {(u, v): sent for u, v in zip(path, path[1:])}
    return flows, sent


def greedy(G):
    """Cheapest surviving route first, push as much as it can carry, then the next route."""
    cap_node = {n: G.nodes[n]["cap"] for n in G.nodes}
    cap_edge = {e: G.edges[e]["cap"] for e in G.edges}
    flows, delivered = {}, 0.0
    while delivered < DEMAND - 1e-9:
        H = nx.DiGraph()
        for (u, v) in G.edges:
            ok = cap_edge[(u, v)] > 1e-9 and cap_node[u] > 1e-9 and cap_node[v] > 1e-9
            if ok:
                H.add_edge(u, v, cost=G.edges[(u, v)]["cost"])
        if SRC not in H or DST not in H or not nx.has_path(H, SRC, DST):
            break
        path = nx.shortest_path(H, SRC, DST, weight="cost")
        edges = list(zip(path, path[1:]))
        push = min([DEMAND - delivered]
                   + [cap_edge[e] for e in edges]
                   + [cap_node[n] for n in path[1:-1]])
        for e in edges:
            cap_edge[e] -= push
            flows[e] = flows.get(e, 0) + push
        for n in path[1:-1]:
            cap_node[n] -= push
        delivered += push
    return flows, delivered


# ---------------------------------------------------------------- experiment
if __name__ == "__main__":
    G0 = build_toy_graph()
    rows = []
    for node in ["A", "B", "C", "D", "E", "F"]:
        G1 = fail_node(G0, node, remaining=0.0)
        for name, (flows, delivered) in {
            "do nothing": do_nothing(G0, G1),
            "greedy": greedy(G1),
            "optimizer": solve_lp(G1),
        }.items():
            r = evaluate(G1, flows, delivered)
            rows.append({"failed hub": node, "plan": name, **r})
    table = pd.DataFrame(rows)
    print(table.pivot(index="failed hub", columns="plan", values="total_cost"))
    print("\nAverage total cost per plan:")
    print(table.groupby("plan")["total_cost"].mean().round(1).sort_values())
    Path("experiments").mkdir(exist_ok=True)
    table.to_csv("experiments/c_toy_results.csv", index=False)
    print("\nSaved experiments/c_toy_results.csv")
